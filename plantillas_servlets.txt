PLANTILLA 3 - Servlet que PROCESA un formulario (crear/registrar) -> doPost
Usalo cuando el enunciado dice: "agregar", "registrar", "crear"
Incluye el patron de validar reglas de negocio y mostrar alertas HTML

package com.exam.app.servlet;

import jakarta.servlet.ServletContext;
import jakarta.servlet.ServletException;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.web.context.WebApplicationContext;
import org.springframework.web.context.support.WebApplicationContextUtils;

import java.io.IOException;
import java.io.PrintWriter;

// TODO: cambiar la ruta (debe coincidir con el "action" del formulario)
@WebServlet("/procesar-algo")
public class ProcesarAlgoServlet extends HttpServlet {

    private MiEntidadService service;

    @Override
    public void init() throws ServletException {
        super.init();
        ServletContext servletContext = getServletContext();
        WebApplicationContext ctx = WebApplicationContextUtils.getWebApplicationContext(servletContext);
        this.service = ctx.getBean(MiEntidadService.class);
    }

    @Override
    protected void doPost(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
        resp.setContentType("text/html;charset=UTF-8");

        try (PrintWriter out = resp.getWriter()) {
            try {
                // TODO: leer y parsear los parametros reales del formulario
                String campo1 = req.getParameter("campo1");
                double campo2 = Double.parseDouble(req.getParameter("campo2"));

                // TODO: llamar al metodo del service que valida las reglas de negocio y guarda
                service.registrar(campo1, campo2);

                out.println("<!DOCTYPE html><html><body>");
                out.println("<h1>Registrado correctamente</h1>");
                out.println("<a href='formulario-algo'>Volver</a>");
                out.println("</body></html>");

            } catch (NumberFormatException e) {
                // parametros invalidos / vacios
                out.println("<!DOCTYPE html><html><body>");
                out.println("<h1>Error</h1><p>Parametros invalidos.</p>");
                out.println("<a href='formulario-algo'>Volver a intentar</a>");
                out.println("</body></html>");

            } catch (RuntimeException e) {
                // TODO: aqui caen las excepciones de reglas de negocio que lance el Service
                // ej: OutOfRangeException, DuplicateTimestampException, etc.
                out.println("<!DOCTYPE html><html><body>");
                out.println("<h1>Alerta</h1><p>" + e.getMessage() + "</p>");
                out.println("<a href='formulario-algo'>Volver a intentar</a>");
                out.println("</body></html>");
            }
        }
    }

    // Opcional: si quieres que la misma URL tambien muestre el formulario por GET
    @Override
    protected void doGet(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
        // TODO: redirigir o reusar la plantilla 2 si el enunciado lo pide
        resp.sendRedirect("formulario-algo");
    }
}

        // TODO: obtener el id desde el query string, ej: /editar-algo?id=3

        Object entidad = service.buscarPorId(id); // TODO: metodo real del service

PLANTILLA 7 - Servlet que reenvia (forward) a un JSP en vez de escribir HTML
Usalo si el profesor pide trabajar con archivos .jsp en webapp/

package com.exam.app.servlet;

import jakarta.servlet.RequestDispatcher;
import jakarta.servlet.ServletContext;
import jakarta.servlet.ServletException;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.web.context.WebApplicationContext;
import org.springframework.web.context.support.WebApplicationContextUtils;

import java.io.IOException;
import java.util.Collection;

// TODO: cambiar la ruta
@WebServlet("/listar-jsp")
public class ListarJspServlet extends HttpServlet {

    private MiEntidadService service;

    @Override
    public void init() throws ServletException {
        WebApplicationContext ctx = WebApplicationContextUtils.getWebApplicationContext(getServletContext());
        this.service = ctx.getBean(MiEntidadService.class);
    }

    @Override
    protected void doGet(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
        Collection<?> datos = service.obtenerTodos();

        // request.setAttribute deja el dato disponible para que el JSP lo lea con ${nombre}
        req.setAttribute("datos", datos);

        // TODO: cambiar la ruta del jsp (normalmente dentro de /WEB-INF/views/ o directo en webapp/)
        RequestDispatcher dispatcher = req.getRequestDispatcher("/listado.jsp");
        dispatcher.forward(req, resp);
    }
}

NOTAS RAPIDAS PARA DECIDIR QUE PLANTILLA USAR
------------------------------------------------------------------------
- El enunciado dice "mostrar/listar/consultar/ver"        -> Plantilla 1
- El enunciado dice "formulario para..."                  -> Plantilla 2
- El enunciado dice "agregar/registrar/crear"              -> Plantilla 2 (form) + 3 (procesa)
- El enunciado dice "editar/actualizar/modificar"          -> Plantilla 4
- El enunciado dice "eliminar/borrar"                      -> Plantilla 5
- El profesor usa .jsp en vez de HTML escrito a mano       -> Plantilla 7
- El profesor pide getBean por nombre (no por clase)       -> Plantilla 6

RECORDATORIO:
- doGet = leer/mostrar, sin efectos secundarios
- doPost = crear/modificar datos, viene de un <form method="post">
- init() solo hace falta si el servlet necesita un bean de Spring
- La ruta del @WebServlet debe coincidir EXACTO con el href/action del HTML que lo llama
